55 // Walk up looking for an anchor: an enclosing body whose canonical
56 // is already known, either because slang set it via setCanonicalBody
57 // (the outermost non-canonical instance) or because it was cached on
58 // a previous query.
59 ast::InstanceBodySymbol const *cur = &body;
60 ast::InstanceBodySymbol const *anchor = nullptr;
61 ast::InstanceBodySymbol const *anchorCanonical = nullptr;
62 while (cur != nullptr) {
63 if (auto it = bodyCache.find(cur); it != bodyCache.end()) {
64 anchor = cur;
65 anchorCanonical = it->second;
66 break;
67 }
68 if (cur->parentInstance != nullptr) {
69 if (auto const *direct = cur->parentInstance->getCanonicalBody();
70 direct != nullptr && direct != cur) {
71 anchor = cur;
72 anchorCanonical = direct;
73 break;
74 }
75 }
76 auto const *parentScope = cur->parentInstance != nullptr
77 ? cur->parentInstance->getParentScope()
78 : nullptr;
79 if (parentScope == nullptr) {
80 break;
81 }
82 cur = parentScope->getContainingInstance();
83 }
84
85 if (anchor == nullptr || anchorCanonical == anchor) {
86 // No redirect along the chain; the body is canonical.
87 bodyCache.emplace(&body, &body);
88 return body;
89 }

92 // register every nested body and value pair.
93 bodyCache.emplace(anchor, anchorCanonical);
94 populatePairedBodies(*anchor, *anchorCanonical);

96 if (auto it = bodyCache.find(&body); it != bodyCache.end()) {
97 return *it->second;
98 }
99 // Defensive: fall back to identity rather than asserting.
100 bodyCache.emplace(&body, &body);

110 void populatePairedBodies(ast::Scope const &local,
111 ast::Scope const &canonical) {
112 auto localIt = local.members().begin();
113 auto localEnd = local.members().end();
114 auto canonIt = canonical.members().begin();
115 auto canonEnd = canonical.members().end();
116 for (; localIt != localEnd && canonIt != canonEnd; ++localIt, ++canonIt) {
117 if (localIt->isValue() && canonIt->isValue()) {
118 valueCache.emplace(&localIt->as<ast::ValueSymbol>(),
119 &canonIt->as<ast::ValueSymbol>());
120 continue;
121 }
122 if (localIt->kind == ast::SymbolKind::Instance &&
123 canonIt->kind == ast::SymbolKind::Instance) {
124 auto const &localInst = localIt->as<ast::InstanceSymbol>();
125 auto const &canonInst = canonIt->as<ast::InstanceSymbol>();
126 auto const *canon = canonInst.getCanonicalBody();
127 auto const &canonBody = canon != nullptr ? *canon : canonInst.body;
128 if (&localInst.body != &canonBody) {
129 bodyCache.emplace(&localInst.body, &canonBody);
130 populatePairedBodies(localInst.body, canonBody);
131 } else {
132 bodyCache.emplace(&localInst.body, &localInst.body);
133 }
134 continue;
135 }
136 if (localIt->kind == canonIt->kind) {
137 if (localIt->kind == ast::SymbolKind::GenerateBlock) {
138 populatePairedBodies(localIt->as<ast::GenerateBlockSymbol>(),

140 } else if (localIt->kind == ast::SymbolKind::GenerateBlockArray) {
141 populatePairedBodies(localIt->as<ast::GenerateBlockArraySymbol>(),
142 canonIt->as<ast::GenerateBlockArraySymbol>());

144 // Array of instances (e.g. `sub u[4](...)`); recurse so each
